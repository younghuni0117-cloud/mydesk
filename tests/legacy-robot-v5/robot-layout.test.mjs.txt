import test from 'node:test';import assert from 'node:assert/strict';import {readFileSync} from 'node:fs';import {runInNewContext} from 'node:vm';
const script=readFileSync(new URL('../game-layout.js',import.meta.url),'utf8');
function layout(width,height,page='robot',scale=1){const classes=new Set(),styles={};const node=()=>({addEventListener(){},focus(){}});const body={dataset:{page},classList:{contains:k=>classes.has(k),toggle(k,on){if(on)classes.add(k);else classes.delete(k);}},style:{setProperty:(k,v)=>styles[k]=v}};const document={body,documentElement:{clientWidth:width},querySelector:()=>null,getElementById:id=>id==='board'?null:node(),addEventListener(){},fullscreenEnabled:false};runInNewContext(script,{document,window:{innerHeight:height,visualViewport:{height,scale,addEventListener(){}},matchMedia:()=>({matches:true,addEventListener(){}}),addEventListener(){}},location:{hash:''},requestAnimationFrame(){return 1},Event:class{}});return {classes,styles};}
test('cover keeps scrolling, inner landscape fits without changing the other games',()=>{const cover=layout(360,780);assert.equal(cover.classes.has('robot-wide-fit'),false);assert.equal(cover.classes.has('fit-game'),false);for(const [w,h]of [[900,650],[800,560],[720,480]])assert.equal(layout(w,h).classes.has('robot-wide-fit'),true);assert.equal(layout(650,900).classes.has('robot-wide-fit'),true);assert.equal(layout(900,650,'robot',1.3).classes.has('robot-wide-fit'),false);assert.equal(layout(360,780,'maze').classes.has('fit-game'),true);assert.equal(layout(900,650,'cartpole').classes.has('fit-game'),true);});
