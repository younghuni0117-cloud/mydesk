import test from 'node:test';import assert from 'node:assert/strict';
import {normalizeEvaluationSettings,monitorCases} from '../robot-evaluation.mjs';
import {RobotLearner} from '../robot-learning.mjs';
import {POLICY_SIZE,evaluateAbility,passedStage,DT} from '../robot-core.mjs';
import {ParallelRollouts} from '../robot-rollouts.mjs';
const zero=Array(POLICY_SIZE).fill(0);
test('evaluation settings default to 10/30 and keep longer tests valid without changing observation time',()=>{assert.deepEqual(normalizeEvaluationSettings(),{validationSeconds:10,monitorSeconds:30});assert.deepEqual(normalizeEvaluationSettings({validationSeconds:16,monitorSeconds:45,observationSeconds:1}),{validationSeconds:16,monitorSeconds:45});assert.deepEqual(normalizeEvaluationSettings({validationSeconds:-1,monitorSeconds:NaN}),{validationSeconds:10,monitorSeconds:30});assert.equal(normalizeEvaluationSettings({validationSeconds:40,monitorSeconds:30}).monitorSeconds,40);});
test('held-out pushes cover the whole extra test, with reproducible diverse times, forces and directions',()=>{const a=monitorCases(30),b=monitorCases(60);assert.deepEqual(a,monitorCases(30));assert.equal(a.length,2);for(const c of a){assert.equal(c.seconds,30);assert.ok(c.pushes.length>=6);assert.ok(c.pushes.at(-1).time>20);assert.ok(c.pushes.every(p=>p.time>0&&p.time<30&&p.strength>=2&&p.strength<=6));assert.ok(new Set(c.pushes.map(p=>Math.floor(p.direction/90))).size>=2);assert.ok(new Set(c.pushes.map(p=>p.location)).size>=2);}assert.ok(b[0].pushes.length>a[0].pushes.length);});
test('serial and parallel evaluators use the same configured lengths and full-duration pass criteria',async()=>{const env={validationSeconds:6,monitorSeconds:12,observationSeconds:1},pool=new ParallelRollouts({enabled:false});const short=evaluateAbility(zero,'balance',env);assert.ok(short.cases.every(c=>c.metrics.plannedDuration===6&&Math.abs(c.metrics.time-6)<DT));assert.deepEqual(await pool.ability(zero,'balance',env),short);const cases=monitorCases(12),long=evaluateAbility(zero,'balance',env,cases);assert.ok(long.cases.every(c=>c.metrics.plannedDuration===12));assert.deepEqual(await pool.ability(zero,'balance',env,()=>true,cases),long);const metrics={terminationReason:'duration',done:true,time:6,upright:1,falls:0,distance:0,height:1};assert.equal(passedStage(metrics,'balance',6),true);assert.equal(passedStage(metrics,'balance',10),false);pool.close();});
test('changing evaluation times preserves learned weights, optimizer, counts and historical records',()=>{const l=new RobotLearner(null,42,{terrain:'meadow'});l.weights[783]=.12;l.mean[783]=.11;l.attempts=100;l.taskAttempts.balance=100;l.algorithmAttempts.ppo=100;const saved=l.snapshot(),raw=JSON.stringify(saved),next=new RobotLearner(saved,42,{terrain:'meadow',validationSeconds:12,monitorSeconds:45});assert.deepEqual(next.weights,saved.weights);assert.deepEqual(next.mean,saved.mean);assert.equal(next.attempts,100);assert.deepEqual(next.taskAttempts,saved.taskAttempts);assert.ok(next.history.some(r=>r.environment===JSON.stringify(saved.environment)));assert.equal(JSON.stringify(saved),raw);assert.ok(RobotLearner.valid(next.snapshot()));assert.ok(next.abilities.balance.cases.every(c=>c.metrics.plannedDuration===12));});
test('only adopted improvements run the longer monitor; monitoring never changes training counts',async()=>{const l=new RobotLearner(null,42,{validationSeconds:6,monitorSeconds:12});const ability=structuredClone(l.abilities.balance);ability.passRate=1;for(const c of ability.cases){c.passed=true;c.metrics.upright=1;c.metrics.falls=0;}const candidate=[...zero];candidate[783]=.01;l.validationBank={weights:candidate,task:'balance',ability};let monitors=0;const validation={ability:async(w,task,env,active,cases)=>{if(cases){monitors++;assert.ok(cases.every(c=>c.seconds===12));return evaluateAbility(w,task,env,cases);}return structuredClone(l.abilities[task]);},commands:async()=>l.goalReward};assert.equal(await l.checkpointAsync(undefined,undefined,validation,true),true);assert.equal(monitors,1);assert.equal(l.generalization.seconds,12);assert.equal(l.attempts,0);assert.deepEqual(l.weights,candidate);await l.checkpointAsync(undefined,undefined,validation,true);assert.equal(monitors,1);const bad=new RobotLearner(null,42,{validationSeconds:6,monitorSeconds:12});bad.abilities.walk.cases[0].passed=true;bad.validationBank={weights:candidate,task:'balance',ability};const rejected={...validation,ability:async(w,task)=>{const a=structuredClone(bad.abilities[task]);a.cases[0].passed=false;return a;}};assert.equal(await bad.checkpointAsync(undefined,undefined,rejected,true),false);assert.equal(monitors,1);assert.equal(bad.generalization,null);});
