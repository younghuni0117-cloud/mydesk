import test from 'node:test';import assert from 'node:assert/strict';import {Worker as NativeWorker} from 'node:worker_threads';
import {RobotLearner} from '../robot-learning.mjs';import {POLICY_SIZE} from '../robot-policy.mjs';import {seeded} from '../robot-core.mjs';import {PPOOptimizer,ppoEpisode} from '../robot-ppo.mjs';import {ParallelRollouts,executeTrial} from '../robot-rollouts.mjs';
const clone=l=>{const {random,...state}=l;return Object.assign(Object.create(RobotLearner.prototype),structuredClone(state),{random:seeded(42)});};
class PhysicsWorker{constructor(){this.ready=false;this.queue=[];this.native=new NativeWorker(new URL('./robot-rollout-harness.mjs',import.meta.url));this.native.on('message',data=>{if(data.type==='wrapper-ready'){this.ready=true;for(const m of this.queue)this.native.postMessage(m);this.queue=[];}else this.onmessage?.({data});});this.native.on('error',()=>this.onerror?.({preventDefault(){}}));}postMessage(data){this.ready?this.native.postMessage(data):this.queue.push(data);}terminate(){return this.native.terminate();}}
test('two physics threads preserve serial trial order, CEM distribution, all validation cases and model weights',async()=>{const pool=new ParallelRollouts({enabled:true,factory:()=>new PhysicsWorker()});try{assert.equal(await pool.start(),true);const baseline=new RobotLearner(null,42);baseline.setAlgorithm('cem');const serial=clone(baseline),parallel=clone(baseline),expected=[],actual=[];for(let i=0;i<32;i++)expected.push(serial.trainOne());await serial.checkpointAsync();for(let i=0;i<32;i+=2){const a=parallel.prepareTrial({record:i===0},0),b=parallel.prepareTrial({},1),[ta,tb]=await pool.pair(a,b);const first=parallel.completeTrial(a,ta),second=parallel.completeTrial(b,tb);actual.push(first,second);if(i===0)assert.ok(first.scene.trace.length>0);}await parallel.checkpointAsync(undefined,undefined,pool);assert.deepEqual(actual.map(t=>t.score),expected.map(t=>t.score));assert.deepEqual(actual.map(t=>t.weights),expected.map(t=>t.weights));for(const key of ['weights','mean','spread','attempts','taskAttempts','policyVersion','abilities','goalReward'])assert.deepEqual(parallel[key],serial[key],key);assert.ok(RobotLearner.valid(parallel.snapshot()));for(const stage of Object.keys(baseline.abilities))assert.deepEqual(await pool.ability(baseline.weights,stage),baseline.abilities[stage]);assert.equal(await pool.commands(baseline.weights),baseline.goalReward);
 const prepared=parallel.prepareTrial({},1),attempts=parallel.attempts;assert.throws(()=>parallel.completeTrial(prepared,executeTrial(prepared)),/순서/);assert.equal(parallel.attempts,attempts);
}finally{pool.close();}});
test('unsupported workers and runtime failures fall back to the exact same episode without losing a result',async()=>{const prepared={weights:Array(POLICY_SIZE).fill(.01),task:'balance',seed:31,options:{summaryOnly:true,autoPush:false}},expected=executeTrial(prepared),disabled=new ParallelRollouts({enabled:false}),blocked=new ParallelRollouts({enabled:true,factory(){throw Error('blocked');}});assert.equal(await disabled.start(),false);assert.equal(await blocked.start(),false);assert.deepEqual(await disabled.run(prepared),expected);let prevented=false;const failing=new ParallelRollouts({enabled:true,factory:()=>{const worker={terminate(){},postMessage(){worker.onerror({preventDefault(){prevented=true;}});}};queueMicrotask(()=>worker.onmessage({data:{type:'ready'}}));return worker;}});try{assert.equal(await failing.start(),true);assert.deepEqual(await failing.run(prepared),expected);assert.equal(failing.ready,false);assert.equal(prevented,true);assert.deepEqual(await failing.run(prepared),expected);}finally{disabled.close();blocked.close();failing.close();}});

test('PPO side worker preserves stochastic actions, log probabilities and experience before the shared update',async()=>{const pool=new ParallelRollouts({enabled:true,factory:()=>new PhysicsWorker()});try{assert.equal(await pool.start(),true);const optimizer=new PPOOptimizer(Array(POLICY_SIZE).fill(0),42),prepared={weights:optimizer.actor,critic:optimizer.critic,logStd:optimizer.logStd,task:'balance',algorithm:'ppo',seed:100,noiseSeed:789,options:{autoPush:true,pushMin:1,pushMax:2,summaryOnly:true}};const expected=ppoEpisode(prepared),actual=await pool.run(prepared);assert.deepEqual(actual.metrics,expected.metrics);assert.deepEqual(actual.experience,expected.experience);assert.deepEqual(actual.pushPlan,expected.pushPlan);}finally{pool.close();}});
